3.2:
	Write Node* insertAtTail(Node* head, int x)
	```
	Node* insertAtTail(Node* head, int x) {
		Node* newNode = new Node(x);
		if (head == NULL) return newNode;
		Node* cur = head;
		while (cur->next != NULL) {
			cur = cur->next;
		}
		cur->next = newNode;
		return head;
	}
	```

3.3:
	Write int indexOf(Node* head, int key) that returns the 0-based position of the first node containing key, or -1 if absent.
	```
	int indexOf(Node* head, int key) {
		int pos = 0;
		if (head == NULL) return -1;
		Node* cur = head;
		while (cur->next != NULL) {
			if (cur->data == key) return pos;
			cur = cur->next;
			pos++;
		}
		return -1;
	}
	```
3.4:
	The following code is meant to insert 7 at the head. Find the bug and fix it.
	```
	Node* n = new Node(7);
	head = n;
	n->next = head;
	```
	head = n is done before n->next is done, so head-next points to head.
	fix
	```
	Node* n = new Node(7);
	n->next = head;
	head = n;
	```

3.5:
	Write Node* deleteLast() that removes the last node and returns the head.
	```
	Node* deleteLast() {
		if (head == NULL) return NULL;
		if (head->next == NULL) {
			delete head;
			return NULL;
		}
		Node* cur = head;
		while (cur->next->next != NULL) {
			cur = cur->next;
		}
		delete cur->next;
		cur->next == NULL;
		return head;
	}
	```

3.6:
	Write Node* deleteAt(int k) that deletes node at 0-based position k. if k is out of range the list stays unchanged.
	```
	Node* deleteAt(int k) ,
		if (head == NULL || k < 0) return head;
		if (k == 0) {
			Node* t = head;
			head = head->next;
			delete t;
			return head;
		}
		Node* cur = head;
		for (int i = 0; i < k - 1 && cur->next != NULL; i++) {
			cur = cur->next;
		}
		if (cur->next == NULL) return head;
		Node* t = cur->next;
		cur->next = t->next;
		delete t;
		return head;
	}
	```

4.3:
  Linked-list stack. Write void push(int x) and int pop() (print "Underflow" and return -1 if empty). The stack top is the head of the list.
  ```
  void push(int x) {
      Node* n = new Node(x);
      n->next = *head;
      *head = n;
    }

  int pop() {
      if (*head == NULL) return -1;
      Node *h = *head;
      int v = h->data;
      *head = h->next;
      delete h;
      return v;
    }
    ```

4.4:
  In a singly linked-list stack, why do we use the head of the list as the top rather than the last node?
  Because inserting at head for a singly linked-list is O(1) which is faster than the O(n) at the tail.
